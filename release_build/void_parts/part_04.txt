            (int, float)
        ):
            continue

        stapel = max(
            1,
            aktiv_daten.get(
                "stapel",
                1
            )
        )

        gesamt += (
            modifikator * stapel
        )

    return gesamt


def status_dauer_verringern(
    einheit,
    menge=1,
    meldung=True
):
    if (
        isinstance(menge, bool)
        or not isinstance(
            menge,
            (int, float)
        )
        or menge <= 0
    ):
        return []

    aktive = status_effekte_aktiv()

    beendet = []

    for status_id, aktiv_daten in list(
        aktive.items()
    ):
        if aktiv_daten.get(
            "einheit"
        ) != einheit:
            continue

        dauer = aktiv_daten.get(
            "dauer"
        )

        if not isinstance(
            dauer,
            (int, float)
        ):
            continue

        aktiv_daten[
            "dauer"
        ] = dauer - menge

        if aktiv_daten[
            "dauer"
        ] <= 0:
            beendet.append(
                status_id
            )

    for status_id in beendet:
        status_entfernen(
            status_id,
            meldung=meldung
        )

    return beendet

def status_schritte_abschliessen(
    menge=1
):
    if (
        isinstance(menge, bool)
        or not isinstance(menge, int)
        or menge <= 0
    ):
        return True

    for _ in range(menge):
        aktive = status_effekte_aktiv()

        for status_id, aktiv_daten in list(
            aktive.items()
        ):
            if aktiv_daten.get(
                "einheit"
            ) != "schritte":
                continue

            definition = STATUS_EFFEKTE.get(
                status_id,
                {}
            )

            if not definition.get(
                "schaden_ausserhalb_kampf",
                False
            ):
                continue

            tick_schaden = definition.get(
                "schaden_pro_tick",
                0
            )

            if not isinstance(
                tick_schaden,
                (int, float)
            ):
                continue

            stapel = max(
                1,
                aktiv_daten.get(
                    "stapel",
                    1
                )
            )

            schaden = int(
                tick_schaden * stapel
            )

            if schaden <= 0:
                continue

            spieler[
                "lebenspunkte"
            ] = max(
                0,
                spieler[
                    "lebenspunkte"
                ] - schaden
            )

            name = definition.get(
                "name",
                status_id
            )

            print()

            design_text(
                (
                    f"{name}: Du erleidest "
                    f"{schaden} Schaden."
                ),
                design="gefahr",
                art="menue"
            )

        if spieler[
            "lebenspunkte"
        ] <= 0:
            print()

            design_text(
                "Du brichst an deinen Verletzungen zusammen.",
                design="gefahr",
                art="menue"
            )

            spieler[
                "spiel_laeuft"
            ] = False

            return False

        status_dauer_verringern(
            "schritte"
        )

    return True

def attribut_system_sicherstellen(
    spieler_daten=None
):
    if spieler_daten is None:
        spieler_daten = globals().get(
            "spieler"
        )

    if not isinstance(
        spieler_daten,
        dict
    ):
        return False

    # -------------------------
    # HAUPTATTRIBUTE
    # -------------------------

    attribute = spieler_daten.get(
        "attribute"
    )

    if not isinstance(
        attribute,
        dict
    ):
        attribute = {}

        spieler_daten[
            "attribute"
        ] = attribute

    for attribut_id, definition in (
        ATTRIBUT_DEFINITIONEN.items()
    ):
        wert = attribute.get(
            attribut_id
        )

        if (
            isinstance(
                wert,
                bool
            )
            or not isinstance(
                wert,
                int
            )
        ):
            attribute[
                attribut_id
            ] = definition[
                "startwert"
            ]

    # -------------------------
    # ATTRIBUTFORTSCHRITT
    # -------------------------

    fortschritt = spieler_daten.get(
        "attribut_fortschritt"
    )

    if not isinstance(
        fortschritt,
        dict
    ):
        fortschritt = {}

        spieler_daten[
            "attribut_fortschritt"
        ] = fortschritt

    for attribut_id in (
        ATTRIBUT_DEFINITIONEN
    ):
        wert = fortschritt.get(
            attribut_id
        )

        if (
            isinstance(
                wert,
                bool
            )
            or not isinstance(
                wert,
                (int, float)
            )
        ):
            fortschritt[
                attribut_id
            ] = 0

    # -------------------------
    # VERFASSUNGEN
    # -------------------------

    verfassungen = spieler_daten.get(
        "verfassungen"
    )

    if not isinstance(
        verfassungen,
        dict
    ):
        verfassungen = {}

        spieler_daten[
            "verfassungen"
        ] = verfassungen

    praegung = spieler_daten.get(
        "verfassungs_praegung"
    )

    if not isinstance(
        praegung,
        dict
    ):
        praegung = {}

        spieler_daten[
            "verfassungs_praegung"
        ] = praegung

    for verfassung_id in (
        VERFASSUNG_DEFINITIONEN
    ):
        wert = verfassungen.get(
            verfassung_id
        )

        if (
            isinstance(
                wert,
                bool
            )
            or not isinstance(
                wert,
                int
            )
        ):
            verfassungen[
                verfassung_id
            ] = 0

        wert = praegung.get(
            verfassung_id
        )

        if (
            isinstance(
                wert,
                bool
            )
            or not isinstance(
                wert,
                (int, float)
            )
        ):
            praegung[
                verfassung_id
            ] = 0

    # -------------------------
    # LERNQUELLEN
    # -------------------------

    lernquellen = spieler_daten.get(
        "lernquellen"
    )

    if not isinstance(
        lernquellen,
        dict
    ):
        spieler_daten[
            "lernquellen"
        ] = {}

    return True

def verfassung_wert_aus_praegung(
    punkte
):
    if (
        isinstance(
            punkte,
            bool
        )
        or not isinstance(
            punkte,
            (int, float)
        )
    ):
        return 0

    if punkte >= 18:
        return 3

    if punkte >= 10:
        return 2

    if punkte >= 4:
        return 1

    return 0


def verfassung_neuberechnen(
    verfassung_id,
    meldung=True
):
    attribut_system_sicherstellen()

    if (
        verfassung_id
        not in VERFASSUNG_DEFINITIONEN
    ):
        return False

    praegung = spieler[
        "verfassungs_praegung"
    ].get(
        verfassung_id,
        0
    )

    alter_wert = spieler[
        "verfassungen"
    ].get(
        verfassung_id,
        0
    )

    neuer_wert = (
        verfassung_wert_aus_praegung(
            praegung
        )
    )

    if neuer_wert == alter_wert:
        return False

    spieler[
        "verfassungen"
    ][
        verfassung_id
    ] = neuer_wert

    if (
        meldung
        and neuer_wert > alter_wert
    ):
        print()

        design_text(
            "VERFASSUNG ENTWICKELT",
            design="positiv",
            art="menue"
        )

        print(
            f"{VERFASSUNG_DEFINITIONEN[verfassung_id]['name']}: "
            f"+{alter_wert} → +{neuer_wert}"
        )

    return True

def attribut_schwelle(
    attribut_id
):
    attribut_system_sicherstellen()

    if attribut_id not in ATTRIBUT_DEFINITIONEN:
        return None

    aktueller_wert = spieler[
        "attribute"
    ][
        attribut_id
    ]

    # Das Übersinnliche muss zunächst
    # überhaupt erst erwachen.
    if (
        attribut_id
        == "uebersinnliches"
        and aktueller_wert == 0
    ):
        return 6

    return (
        8
        + aktueller_wert * 4
    )


def attribut_fortschritt_hinzufuegen(
    attribut_id,
    menge=1,
    praegungen=None,
    meldung=True
):
    attribut_system_sicherstellen()

    if attribut_id not in ATTRIBUT_DEFINITIONEN:
        return False

    if (
        isinstance(
            menge,
            bool
        )
        or not isinstance(
            menge,
            (int, float)
        )
        or menge <= 0
    ):
        return False

    if praegungen is None:
        praegungen = {}

    if not isinstance(
        praegungen,
        dict
    ):
        return False

    # -------------------------
    # VERFASSUNGSPRÄGUNG
    # -------------------------

    for (
        verfassung_id,
        praegung_menge
    ) in praegungen.items():

        definition = (
            VERFASSUNG_DEFINITIONEN.get(
                verfassung_id
            )
        )

        if not definition:
            continue

        # Eine Stärke-Prägung darf z.B.
        # nicht Geist trainieren.
        if (
            definition[
                "attribut"
            ] != attribut_id
        ):
            continue

        if (
            isinstance(
                praegung_menge,
                bool
            )
            or not isinstance(
                praegung_menge,
                (int, float)
            )
            or praegung_menge <= 0
        ):
            continue

        spieler[
            "verfassungs_praegung"
        ][
            verfassung_id
        ] += praegung_menge

        verfassung_neuberechnen(
            verfassung_id,
            meldung=meldung
        )

    # -------------------------
    # HAUPTATTRIBUT
    # -------------------------

    spieler[
        "attribut_fortschritt"
    ][
        attribut_id
    ] += menge

    entwickelt = False

    while True:
        schwelle = attribut_schwelle(
            attribut_id
        )

        if schwelle is None:
            break

        fortschritt = spieler[
            "attribut_fortschritt"
        ][
            attribut_id
        ]

        if fortschritt < schwelle:
            break

        spieler[
            "attribut_fortschritt"
        ][
            attribut_id
        ] -= schwelle

        alter_wert = spieler[
            "attribute"
        ][
            attribut_id
        ]

        spieler[
            "attribute"
        ][
            attribut_id
        ] += 1

        neuer_wert = spieler[
            "attribute"
        ][
            attribut_id
        ]

        entwickelt = True

        if meldung:
            print()

            design_text(
                "ATTRIBUT ENTWICKELT",
                design="positiv",
                art="menue"
            )

            print()

            print(
                f"{ATTRIBUT_DEFINITIONEN[attribut_id]['name']}: "
                f"{alter_wert} → {neuer_wert}"
            )

    return entwickelt
