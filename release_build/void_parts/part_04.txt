            (int, float)
        ):
            continue

        stapel = max(
            1,
            aktiv_daten.get(
                "stapel",
                1
            )
        )

        gesamt += (
            modifikator * stapel
        )

    return gesamt


def status_dauer_verringern(
    einheit,
    menge=1,
    meldung=True
):
    if (
        isinstance(menge, bool)
        or not isinstance(
            menge,
            (int, float)
        )
        or menge <= 0
    ):
        return []

    aktive = status_effekte_aktiv()

    beendet = []

    for status_id, aktiv_daten in list(
        aktive.items()
    ):
        if aktiv_daten.get(
            "einheit"
        ) != einheit:
            continue

        dauer = aktiv_daten.get(
            "dauer"
        )

        if not isinstance(
            dauer,
            (int, float)
        ):
            continue

        aktiv_daten[
            "dauer"
        ] = dauer - menge

        if aktiv_daten[
            "dauer"
        ] <= 0:
            beendet.append(
                status_id
            )

    for status_id in beendet:
        status_entfernen(
            status_id,
            meldung=meldung
        )

    return beendet

def status_schritte_abschliessen(
    menge=1
):
    if (
        isinstance(menge, bool)
        or not isinstance(menge, int)
        or menge <= 0
    ):
        return True

    for _ in range(menge):
        aktive = status_effekte_aktiv()

        for status_id, aktiv_daten in list(
            aktive.items()
        ):
            if aktiv_daten.get(
                "einheit"
            ) != "schritte":
                continue

            definition = STATUS_EFFEKTE.get(
                status_id,
                {}
            )

            if not definition.get(
                "schaden_ausserhalb_kampf",
                False
            ):
                continue

            tick_schaden = definition.get(
                "schaden_pro_tick",
                0
            )

            if not isinstance(
                tick_schaden,
                (int, float)
            ):
                continue

            stapel = max(
                1,
                aktiv_daten.get(
                    "stapel",
                    1
                )
            )

            schaden = int(
                tick_schaden * stapel
            )

            if schaden <= 0:
                continue

            spieler[
                "lebenspunkte"
            ] = max(
                0,
                spieler[
                    "lebenspunkte"
                ] - schaden
            )

            name = definition.get(
                "name",
                status_id
            )

            print()

            design_text(
                (
                    f"{name}: Du erleidest "
                    f"{schaden} Schaden."
                ),
                design="gefahr",
                art="menue"
            )

        if spieler[
            "lebenspunkte"
        ] <= 0:
            print()

            design_text(
                "Du brichst an deinen Verletzungen zusammen.",
                design="gefahr",
                art="menue"
            )

            spieler[
                "spiel_laeuft"
            ] = False

            return False

        status_dauer_verringern(
            "schritte"
        )

    return True

    ):

        )

            not isinstance(
                wert,
                bool
            )
            and isinstance(
                wert,
                (int, float)
            )
        ):
            basis_wert = wert

        wert = spieler.get(

    )


def reise_bedingung_erfuellt(bedingung):
    if not bedingung:
        return True

    if not isinstance(
        bedingung,
        dict
    ):
        return False

    typ = bedingung.get(
        "typ"
    )

    if typ in (
        "alle",
        "eine"
    ):
        bedingungen = bedingung.get(
            "bedingungen",
            []
        )

        if not bedingungen:
            return False

        ergebnisse = [
            reise_bedingung_erfuellt(
                einzelne_bedingung
            )
            for einzelne_bedingung
            in bedingungen
        ]

        if typ == "alle":
            return all(
                ergebnisse
            )

        return any(
            ergebnisse
        )

    if typ == "nicht":
        einzelne_bedingung = bedingung.get(
            "bedingung"
        )

        if not einzelne_bedingung:
            return False

        return not reise_bedingung_erfuellt(
            einzelne_bedingung
        )

    if typ == "besonderer_fund_status":
        return besonderer_fund_status(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "weg_status":
        return weg_status(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "quest_status":
        return spieler.get(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "quest_status_in":
        return spieler.get(
            bedingung.get("id")
        ) in bedingung.get(
            "werte",
            []
        )

    if typ in (
        "spieler_wert_min",
        "spieler_wert_max"
    ):
        wert = spieler_pruefwert(
            bedingung.get("id")
        )

        grenze = bedingung.get(
            "grenze"
        )

        if not isinstance(
            wert,
            (int, float)
        ):
            return False

        if not isinstance(
            grenze,
            (int, float)
        ):
            return False

        if typ == "spieler_wert_min":
            return wert >= grenze

        return wert <= grenze

    return False

def reise_ereignisse_pruefen(
    reise_daten,
    schritt
):
    richtung = (
        f'{reise_daten["start"]}>'
        f'{reise_daten["ziel"]}'
    )

    ausgeloest = reise_daten.setdefault(
        "ausgeloeste_ereignisse",
        []
    )

    dauerhaft_erlebt = spieler.setdefault(
        "reise_ereignisse",
        []
    )

    ereignis_angezeigt = False

    strecken_daten = reise_strecken_daten(
        reise_daten
    )

    for ereignis in strecken_daten.get(
        "ereignisse",
        []
    ):
        ereignis_id = ereignis.get(
            "id"
        )

        if not ereignis_id:
            continue

        if ereignis.get(
            "schritt"
        ) != schritt:
            continue

        ereignis_richtung = ereignis.get(
            "richtung"
        )

        if (
            ereignis_richtung
            and ereignis_richtung != richtung
        ):
            continue

        if ereignis_id in ausgeloest:
            continue

        if (
            ereignis.get(
                "einmalig",
                False
            )
            and ereignis_id
            in dauerhaft_erlebt
        ):
            continue

        if not reise_bedingung_erfuellt(
            ereignis.get(
                "bedingung"
            )
        ):
            continue

        ausgeloest.append(
            ereignis_id
        )

        if (
            ereignis.get(
                "einmalig",
                False
            )
            and ereignis_id
            not in dauerhaft_erlebt
        ):
            dauerhaft_erlebt.append(
                ereignis_id
            )

        aktion = ereignis.get(
            "aktion"
        )

        if aktion == "hilda_karren":
            ereignis_angezeigt = bool(
                hilda_karren_ereignis()
            )

            continue

        print()

        titel = ereignis.get(
            "titel"
        )

        if titel:
            design_text(
                titel,
                design=ereignis.get(
                    "design",
                    "wissen"
                ),
                art="menue"
            )

            print()

        zeilen = ereignis.get(
            "text",
            []
        )

        for nummer, zeile in enumerate(
            zeilen
        ):
            text_ausgeben(
                zeile
            )

            if nummer < len(zeilen) - 1:
                print()

        warte_auf_taste(
            ereignis.get(
                "weiter_text",
                "Weiterreisen"
            )
        )

        ereignis_angezeigt = True

    return ereignis_angezeigt
