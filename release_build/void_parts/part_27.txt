def wald_seitenpfad_navigieren(
    start_knoten="eingang"
):
    if (
        start_knoten
        not in WALD_SEITENPFAD_KNOTEN
    ):
        start_knoten = "eingang"

    aktuelle_position = start_knoten

    while True:
        erstmals = wald_knoten_betreten(
            aktuelle_position
        )

        wald_knoten_beschreibung(
            aktuelle_position,
            erstmals
        )

        # -------------------------
        # ERFOLG: GEHEIMER BACHARM
        # -------------------------

        if (
            aktuelle_position
            == "verborgenes_bachufer"
            and erstmals
        ):
            erfolg_freischalten(
                "wo_das_wasser_schweigt"
            )

        # -------------------------
        # GLÜCKSPROBE AM BACH
        # -------------------------

        if aktuelle_position == "bachufer":
            wald_bach_geheimweg_pruefen()

        aktionen = wald_knoten_aktionen(
            aktuelle_position
        )

        verbindungen = wald_verbindungen(
            aktuelle_position
        )

        if (
            aktuelle_position == "eingang"
            and "hauptweg" not in verbindungen
        ):
            verbindungen.append(
                "hauptweg"
            )

        menue_trenner()

        optionen = []

        # -------------------------
        # AKTIONEN AM ORT
        # -------------------------

        for aktion_id, text in aktionen:
            optionen.append(
                (
                    "aktion",
                    aktion_id
                )
            )

            print(
                f"{len(optionen)}. "
                f"{text}"
            )

        # -------------------------
        # KARTE
        # -------------------------

        optionen.append(
            (
                "karte",
                None
            )
        )

        print(
            f"{len(optionen)}. Karte ansehen"
        )

        # -------------------------
        # WEGE
        # -------------------------

        for ziel_id in verbindungen:
            optionen.append(
                (
                    "weg",
                    ziel_id
                )
            )

            print(
                f"{len(optionen)}. "
                f"{wald_weg_text(
                    aktuelle_position,
                    ziel_id
                )}"
            )

        auswahl = eingabe_menu(
            1,
            len(optionen),
            sonder_tasten=()
        )

        typ, wert = optionen[
            auswahl - 1
        ]

        # -------------------------
        # KARTE
        # -------------------------

        if typ == "karte":
            wald_gebietskarte_anzeigen()
            continue

        # -------------------------
        # ORTSAKTION
        # -------------------------

        if typ == "aktion":
            ergebnis = (
                wald_knoten_aktion_ausfuehren(
                    aktuelle_position,
                    wert
                )
            )

            if ergebnis == "hauptweg":
                daten = wald_erforschung_status()

                daten[
                    "aktueller_knoten"
                ] = None

                return

            continue

        # -------------------------
        # WEG
        # -------------------------

        ziel_id = wert

        if ziel_id == "hauptweg":
            daten = wald_erforschung_status()

            daten[
                "aktueller_knoten"
            ] = None

            return

        ressourcen_vorkommen_regenerieren(
            "wald_holz"
        )

        aktuelle_position = ziel_id

def alter_seitenpfad_hier_verfuegbar():
    reise_daten = spieler.get(
        "reise"
    )

    if not isinstance(
        reise_daten,
        dict
    ):
        return False

    richtung = (
        f'{reise_daten["start"]}>'
        f'{reise_daten["ziel"]}'
    )

    aktueller_schritt = reise_daten.get(
        "aktueller_schritt"
    )

    strecken_daten = reise_strecken_daten(
        reise_daten
    )

    for abzweigung in strecken_daten.get(
        "abzweigungen",
        []
    ):
        if abzweigung.get(
            "id"
        ) != "alter_seitenpfad":
            continue

        ziel_schritt = abzweigung.get(
            "schritt"
        )

        richtungs_schritte = abzweigung.get(
            "schritte",
            {}
        )

        if richtung in richtungs_schritte:
            ziel_schritt = richtungs_schritte[
                richtung
            ]

        return (
            aktueller_schritt
            == ziel_schritt
        )

    return False

def wald_umgebung_erkunden():
    while True:
        ort_titel("Wald - Umgebung")

        optionen = [
            (
                "holz",
                "Nach trockenem Holz suchen"
            )
        ]

        if alter_seitenpfad_hier_verfuegbar():
            seitenpfad_status = weg_status(
                "alter_seitenpfad"
            )

            if seitenpfad_status == "bemerkt":
                optionen.append(
                    (
                        "seitenpfad",
                        "Nach dem bemerkten Seitenpfad suchen"
                    )
                )

            elif seitenpfad_status == "entdeckt":
                optionen.append(
                    (
                        "seitenpfad",
                        "Dem Alten Seitenpfad folgen"
                    )
                )

        for nummer, (_, text) in enumerate(
            optionen,
            start=1
        ):
            print(
                f"{nummer}. {text}"
            )

        zurueck_nummer = len(optionen) + 1

        print(
            f"{zurueck_nummer}. Zurück"
        )

        auswahl = eingabe_menu(
            1,
            zurueck_nummer,
            sonder_tasten=()
        )

        if auswahl == zurueck_nummer:
            return

        aktion = optionen[
            auswahl - 1
        ][0]

        if aktion == "holz":
            holz_sammeln()

        elif aktion == "seitenpfad":
            if weg_status(
                "alter_seitenpfad"
            ) == "entdeckt":
                weg_als_gesehen_markieren(
                    "alter_seitenpfad"
                )

                wald_abzweigung(
                    direkt=True
                )

                return

            wald_abzweigung()
