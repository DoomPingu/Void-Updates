def reise_umkehren():
    reise_daten = spieler.get(
        "reise"
    )

    if not isinstance(
        reise_daten,
        dict
    ):
        return False

    start = reise_daten[
        "start"
    ]

    ziel = reise_daten[
        "ziel"
    ]

    gesamt_schritte = reise_daten[
        "gesamt_schritte"
    ]

    aktueller_schritt = reise_daten[
        "aktueller_schritt"
    ]

    start_name = reise_ort_name(
        start
    )

    ziel_name = reise_ort_name(
        ziel
    )

    ort_titel(
        "Reise umkehren"
    )

    print(
        f"Aktuelles Ziel: {ziel_name}"
    )

    print(
        f"Zurück nach: {start_name}"
    )

    print(
        f"Position: "
        f"{aktueller_schritt}/"
        f"{gesamt_schritte}"
    )

    menue_trenner()

    print(
        "1. Umkehren"
    )

    print(
        f"2. Weiter Richtung "
        f"{ziel_name}"
    )

    auswahl = eingabe_menu(
        1,
        2,
        sonder_tasten=()
    )

    if auswahl == 2:
        return False

    ankunft_texte = (
        reise_daten.setdefault(
            "ankunft_texte",
            {}
        )
    )

    ankunft_texte.setdefault(
        ziel,
        list(
            reise_daten.get(
                "ankunft_zeilen",
                []
            )
        )
    )

    ankunft_texte.setdefault(
        start,
        [
            (
                f"Du erreichst wieder "
                f"{start_name}."
            )
        ]
    )

    # Noch keinen Schritt gegangen:
    # direkt zum Ausgangsort zurück.
    if aktueller_schritt <= 0:
        spieler[
            "reise"
        ] = None

        spieler[
            "ort"
        ] = start

        ort_nachricht_setzen(
            "Du entscheidest dich umzukehren.",
            (
                f"Du kehrst nach "
                f"{start_name} zurück."
            )
        )

        return True

    # Richtung wechseln.
    reise_daten[
        "start"
    ] = ziel

    reise_daten[
        "ziel"
    ] = start

    # Position auf derselben Strecke spiegeln.
    reise_daten[
        "aktueller_schritt"
    ] = (
        gesamt_schritte
        - aktueller_schritt
    )

    # Abzweigungen dürfen auf dem Rückweg
    # erneut angeboten werden.
    reise_daten[
        "angebotene_abzweigungen"
    ] = []

    reise_daten[
        "ankunft_zeilen"
    ] = list(
        ankunft_texte[
            start
        ]
    )

    ort_nachricht_setzen(
        "Du bleibst stehen und kehrst um.",
        (
            f"Du gehst nun zurück in "
            f"Richtung {start_name}."
        )
    )

    return True


def reise_kampfbegegnung_pruefen(
    reise_daten
):
    strecken_daten = (
        reise_strecken_daten(
            reise_daten
        )
    )

    kampf_daten = (
        strecken_daten.get(
            "kampfbegegnungen",
            {}
        )
    )

    if not isinstance(
        kampf_daten,
        dict
    ):
        return False

    if not kampf_daten.get(
        "aktiv",
        False
    ):
        return False

    # Der ruhige Prolog bleibt kampffrei.
    if (
        kampf_daten.get(
            "erst_nach_prolog",
            False
        )
        and spieler.get(
            "prolog_status"
        ) != "prolog_abgeschlossen"
    ):
        return False

    # Ohne Waffe keine zufällige
    # Kampfbegegnung starten.
    if not spieler.get(
        "waffe"
    ):
        return False

    kampf_status = (
        spieler.setdefault(
            "reise_kampf_status",
            {}
        )
    )

    kampf_status.setdefault(
        "schritte_ohne_kampf",
        0
    )

    kampf_status.setdefault(
        "cooldown",
        0
    )

    # Nach einem Kampf einige Schritte Ruhe.
    if kampf_status[
        "cooldown"
    ] > 0:
        kampf_status[
            "cooldown"
        ] -= 1

        return False

    kampf_status[
        "schritte_ohne_kampf"
    ] += 1

    chance = kampf_daten.get(
        "chance",
        0
    )

    garantie_nach = kampf_daten.get(
        "garantie_nach",
        0
    )

    zufall_trifft = (
        isinstance(
            chance,
            (int, float)
        )
        and random.random() < chance
    )

    garantie_trifft = (
        isinstance(
            garantie_nach,
            int
        )
        and garantie_nach > 0
        and kampf_status[
            "schritte_ohne_kampf"
        ] >= garantie_nach
    )

    if not (
        zufall_trifft
        or garantie_trifft
    ):
        return False

    kampf_status[
        "schritte_ohne_kampf"
    ] = 0

    kampf_status[
        "cooldown"
    ] = max(
        0,
        int(
            kampf_daten.get(
                "cooldown",
                0
            )
        )
    )

    ort_titel(
        "Wald - Begegnung"
    )

    text_ausgeben(
        "Ein Geräusch abseits des Weges "
        "lässt dich stehen bleiben."
    )

    print()

    text_ausgeben(
        "Etwas bewegt sich zwischen "
        "den Bäumen."
    )

    print()

    kampf_ergebnis = kampf_starten(
        lebenspunkte=spieler[
            "lebenspunkte"
        ],
        heiltraenke=spieler[
            "heiltränke"
        ],
        waffe=spieler[
            "waffe"
        ]
    )

    spieler[
        "lebenspunkte"
    ] = kampf_ergebnis[
        "lebenspunkte"
    ]

    spieler[
        "heiltränke"
    ] = kampf_ergebnis[
        "heiltraenke"
    ]

    if kampf_ergebnis.get(
        "gewonnen",
        False
    ):
        spieler[
            "besiegte_gegner"
        ] += 1

    elif kampf_ergebnis.get(
        "besiegt",
        False
    ):
        print()

        design_text(
            "Game Over",
            design="gefahr",
            art="menue"
        )

        spieler[
            "spiel_laeuft"
        ] = False

    return True


def reise_fortsetzen():
    global TEXT_SKIP_AKTIV
    reise_daten = spieler.get("reise")

    if not reise_daten:
        return

    start = reise_daten["start"]
    ziel = reise_daten["ziel"]
    gebiet = reise_daten["gebiet"]
    gesamt_schritte = reise_daten["gesamt_schritte"]

    zufallsereignisse = reise_daten.get(
        "zufallsereignisse",
        True
    )

    reise_name = reise_daten.get(
        "name",
        gebiet
    )

    erster_angezeigter_schritt = True

    for schritt in range(
        reise_daten["aktueller_schritt"] + 1,
        gesamt_schritte + 1
    ):
        reise_daten["aktueller_schritt"] = schritt

        TEXT_SKIP_AKTIV = False

        ort_titel(
            gebiet
        )

        if erster_angezeigter_schritt:
            ort_nachricht_anzeigen()
            erster_angezeigter_schritt = False

        print(f"{start} → {ziel}")
        print()

        design_text(
            f"Schritt {schritt}/{gesamt_schritte}",
            design="wissen",
            art="menue"
        )

        menue_trenner()

        richtung = f"{start}>{ziel}"

        strecke = frozenset((start, ziel))

        schritt_texte = REISE_STRECKEN.get(
            strecke,
            {}
        ).get(
            "schritt_texte",
            {}
        ).get(
            richtung,
            []
        )

        print()

        if schritt - 1 < len(schritt_texte):
            text_ausgeben(
                schritt_texte[schritt - 1],
                art="atmosphaere"
            )

        else:
            text_ausgeben(
                f"Du folgst dem Weg weiter in Richtung {ziel}.",
                art="atmosphaere"
            )

        if reise_bekannte_abzweigung_vorhanden(
            reise_daten,
            schritt
        ):
            menue_trenner()

            print("1. Weitergehen")
            print("2. Umgebung erkunden [!]")

            schritt_auswahl = eingabe_menu(
                1,
                2,
                sonder_tasten=()
            )

            if schritt_auswahl == 2:
                if reise_abzweigungen_pruefen(
                    reise_daten,
                    schritt,
                    ziel,
                    erkundung=True
                ):
                    status_schritte_abschliessen()
                    return

                print()

                warte_auf_taste(
                    "Weitergehen"
                )

        else:
            warte_auf_taste(
                "Weitergehen"
            )

        if gebiet in (
            "Wald",
            "Waldweg"
        ):
            ressourcen_vorkommen_regenerieren(
                "wald_holz"
            )

        festes_ereignis = (
            reise_ereignisse_pruefen(
                reise_daten,
                schritt
            )
        )

        kampfbegegnung = False

        if not festes_ereignis:
            kampfbegegnung = (
                reise_kampfbegegnung_pruefen(
                    reise_daten
                )
            )

        if not spieler[
            "spiel_laeuft"
        ]:
            return


        if (
            zufallsereignisse
            and not festes_ereignis
            and not kampfbegegnung
        ):
            spieler["gold"], spieler["heiltränke"], spieler["diamanten"], \
            spieler["lebenspunkte"], spieler["spiel_laeuft"], \
            spieler["besiegte_gegner"], spieler["ort"] = zufallsereignis(
                spieler["gold"],
                spieler["heiltränke"],
                spieler["diamanten"],
                spieler["lebenspunkte"],
                spieler["spiel_laeuft"],
                spieler["besiegte_gegner"],
                spieler["ort"]
            )

            if not spieler["spiel_laeuft"]:
                return

            if spieler["fackel_dauer"] > 0:
                spieler["fackel_dauer"] -= 1

                if spieler["fackel_dauer"] == 0:
                    print()
                    print(
                        "Deine Fackel flackert ein letztes Mal auf "
                        "und erlischt."
                    )
            warte_auf_taste("Weiterreisen")


        if reise_abzweigungen_pruefen(
            reise_daten,
            schritt,
            ziel
        ):
            status_schritte_abschliessen()
            return

        if not status_schritte_abschliessen():
            return

        if schritt < gesamt_schritte:
            return


    ankunft_zeilen = reise_daten["ankunft_zeilen"]

    spieler["reise"] = None
    spieler["ort"] = ziel

    ort_nachricht_setzen(
        *ankunft_zeilen
    )

def npc_variante_waehlen(
    npc,
    schluessel,
    varianten
):
    daten = npc_status(
        npc
    )

    letzte_variante = daten.get(
        schluessel
    )

    moegliche_varianten = [
        variante
        for variante in varianten
        if variante != letzte_variante
    ]

    if not moegliche_varianten:
        moegliche_varianten = varianten

    variante = random.choice(
        moegliche_varianten
    )

    daten[
        schluessel
    ] = variante

    return variante
