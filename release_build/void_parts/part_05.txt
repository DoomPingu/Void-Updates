def spieler_pruefwert(wert_id):
    attribut_system_sicherstellen()

    basis_wert = None

    attribute = spieler.get(
        "attribute",
        {}
    )

    verfassungen = spieler.get(
        "verfassungen",
        {}
    )

    # -------------------------
    # HAUPTATTRIBUT
    # -------------------------

    if wert_id in ATTRIBUT_DEFINITIONEN:
        wert = attribute.get(
            wert_id
        )

        if (
            not isinstance(
                wert,
                bool
            )
            and isinstance(
                wert,
                (int, float)
            )
        ):
            basis_wert = wert

    # -------------------------
    # VERFASSUNG
    # -------------------------

    elif wert_id in VERFASSUNG_DEFINITIONEN:
        attribut_id = (
            VERFASSUNG_DEFINITIONEN[
                wert_id
            ][
                "attribut"
            ]
        )

        attribut_wert = attribute.get(
            attribut_id,
            0
        )

        verfassung_wert = (
            verfassungen.get(
                wert_id,
                0
            )
        )

        if (
            isinstance(
                attribut_wert,
                (int, float)
            )
            and not isinstance(
                attribut_wert,
                bool
            )
            and isinstance(
                verfassung_wert,
                (int, float)
            )
            and not isinstance(
                verfassung_wert,
                bool
            )
        ):
            basis_wert = (
                attribut_wert
                + verfassung_wert
            )

    # -------------------------
    # ALTE / SONSTIGE WERTE
    # -------------------------

    if basis_wert is None:
        wert = spieler.get(
            wert_id
        )

        if (
            not isinstance(
                wert,
                bool
            )
            and isinstance(
                wert,
                (int, float)
            )
        ):
            basis_wert = wert

    status_bonus = status_modifikator(
        wert_id
    )

    if basis_wert is None:
        if status_bonus != 0:
            return status_bonus

        return None

    return (
        basis_wert
        + status_bonus
    )


def reise_bedingung_erfuellt(bedingung):
    if not bedingung:
        return True

    if not isinstance(
        bedingung,
        dict
    ):
        return False

    typ = bedingung.get(
        "typ"
    )

    if typ in (
        "alle",
        "eine"
    ):
        bedingungen = bedingung.get(
            "bedingungen",
            []
        )

        if not bedingungen:
            return False

        ergebnisse = [
            reise_bedingung_erfuellt(
                einzelne_bedingung
            )
            for einzelne_bedingung
            in bedingungen
        ]

        if typ == "alle":
            return all(
                ergebnisse
            )

        return any(
            ergebnisse
        )

    if typ == "nicht":
        einzelne_bedingung = bedingung.get(
            "bedingung"
        )

        if not einzelne_bedingung:
            return False

        return not reise_bedingung_erfuellt(
            einzelne_bedingung
        )

    if typ == "besonderer_fund_status":
        return besonderer_fund_status(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "weg_status":
        return weg_status(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "quest_status":
        return spieler.get(
            bedingung.get("id")
        ) == bedingung.get("status")

    if typ == "quest_status_in":
        return spieler.get(
            bedingung.get("id")
        ) in bedingung.get(
            "werte",
            []
        )

    if typ in (
        "spieler_wert_min",
        "spieler_wert_max"
    ):
        wert = spieler_pruefwert(
            bedingung.get("id")
        )

        grenze = bedingung.get(
            "grenze"
        )

        if not isinstance(
            wert,
            (int, float)
        ):
            return False

        if not isinstance(
            grenze,
            (int, float)
        ):
            return False

        if typ == "spieler_wert_min":
            return wert >= grenze

        return wert <= grenze

    return False

def reise_ereignisse_pruefen(
    reise_daten,
    schritt
):
    richtung = (
        f'{reise_daten["start"]}>'
        f'{reise_daten["ziel"]}'
    )

    ausgeloest = reise_daten.setdefault(
        "ausgeloeste_ereignisse",
        []
    )

    dauerhaft_erlebt = spieler.setdefault(
        "reise_ereignisse",
        []
    )

    ereignis_angezeigt = False

    strecken_daten = reise_strecken_daten(
        reise_daten
    )

    for ereignis in strecken_daten.get(
        "ereignisse",
        []
    ):
        ereignis_id = ereignis.get(
            "id"
        )

        if not ereignis_id:
            continue

        if ereignis.get(
            "schritt"
        ) != schritt:
            continue

        ereignis_richtung = ereignis.get(
            "richtung"
        )

        if (
            ereignis_richtung
            and ereignis_richtung != richtung
        ):
            continue

        if ereignis_id in ausgeloest:
            continue

        if (
            ereignis.get(
                "einmalig",
                False
            )
            and ereignis_id
            in dauerhaft_erlebt
        ):
            continue

        if not reise_bedingung_erfuellt(
            ereignis.get(
                "bedingung"
            )
        ):
            continue

        ausgeloest.append(
            ereignis_id
        )

        if (
            ereignis.get(
                "einmalig",
                False
            )
            and ereignis_id
            not in dauerhaft_erlebt
        ):
            dauerhaft_erlebt.append(
                ereignis_id
            )

        aktion = ereignis.get(
            "aktion"
        )

        if aktion == "hilda_karren":
            ereignis_angezeigt = bool(
                hilda_karren_ereignis()
            )

            continue

        print()

        titel = ereignis.get(
            "titel"
        )

        if titel:
            design_text(
                titel,
                design=ereignis.get(
                    "design",
                    "wissen"
                ),
                art="menue"
            )

            print()

        zeilen = ereignis.get(
            "text",
            []
        )

        for nummer, zeile in enumerate(
            zeilen
        ):
            text_ausgeben(
                zeile
            )

            if nummer < len(zeilen) - 1:
                print()

        warte_auf_taste(
            ereignis.get(
                "weiter_text",
                "Weiterreisen"
            )
        )

        ereignis_angezeigt = True

    return ereignis_angezeigt
