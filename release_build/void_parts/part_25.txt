            von = min(
                start_spalte,
                ziel_spalte
            )

            bis = max(
                start_spalte,
                ziel_spalte
            )

            for spalte in range(
                von + 1,
                bis
            ):
                linie_setzen(
                    start_zeile,
                    spalte,
                    "─"
                )

            return

        # Gleiche Spalte
        if start_spalte == ziel_spalte:
            von = min(
                start_zeile,
                ziel_zeile
            )

            bis = max(
                start_zeile,
                ziel_zeile
            )

            for zeile in range(
                von + 1,
                bis
            ):
                linie_setzen(
                    zeile,
                    start_spalte,
                    "│"
                )

            return

        # Unterschiedliche Höhe und Spalte
        schritt = (
            1
            if ziel_spalte > start_spalte
            else -1
        )

        for spalte in range(
            start_spalte + schritt,
            ziel_spalte,
            schritt
        ):
            linie_setzen(
                start_zeile,
                spalte,
                "─"
            )

        schritt = (
            1
            if ziel_zeile > start_zeile
            else -1
        )

        for zeile in range(
            start_zeile + schritt,
            ziel_zeile,
            schritt
        ):
            linie_setzen(
                zeile,
                ziel_spalte,
                "│"
            )

    # -------------------------
    # SICHTBARE VERBINDUNGEN
    # -------------------------

    gezeichnete_verbindungen = set()

    for start_id in besuchte_raeume:
        for ziel_id in wegstation_verbindungen(
            start_id
        ):
            if (
                start_id
                not in WEGSTATION_KARTENPOSITIONEN
                or ziel_id
                not in WEGSTATION_KARTENPOSITIONEN
            ):
                continue

            if not raum_sichtbar(
                ziel_id
            ):
                continue

            verbindung = frozenset(
                (
                    start_id,
                    ziel_id
                )
            )

            if verbindung in gezeichnete_verbindungen:
                continue

            gezeichnete_verbindungen.add(
                verbindung
            )

            verbindung_zeichnen(
                start_id,
                ziel_id
            )

    # -------------------------
    # RÄUME EINZEICHNEN
    # -------------------------

    for (
        raum_id,
        position
    ) in WEGSTATION_KARTENPOSITIONEN.items():

        text = feld(
            raum_id
        )

        if text is None:
            continue

        zeile, spalte = position

        for index, zeichen in enumerate(
            text
        ):
            x = spalte + index

            if (
                0 <= zeile < hoehe
                and 0 <= x < breite
            ):
                karte[
                    zeile
                ][x] = zeichen

    # -------------------------
    # KARTE AUSGEBEN
    # -------------------------

    ort_titel(
        "Gebietskarte - Alte Wegstation"
    )

    sichtbare_zeilen = [
        index
        for index, zeile in enumerate(karte)
        if any(
            zeichen != " "
            for zeichen in zeile
        )
    ]

    if sichtbare_zeilen:
        erste_zeile = min(
            sichtbare_zeilen
        )

        letzte_zeile = max(
            sichtbare_zeilen
        )

        sichtbarer_bereich = karte[
            erste_zeile:
            letzte_zeile + 1
        ]

        belegte_spalten = []

        for zeile in sichtbarer_bereich:
            for index, zeichen in enumerate(
                zeile
            ):
                if zeichen != " ":
                    belegte_spalten.append(
                        index
                    )

        if belegte_spalten:
            erste_spalte = min(
                belegte_spalten
            )

            letzte_spalte = max(
                belegte_spalten
            )

            for zeile in sichtbarer_bereich:
                print(
                    "  "
                    + "".join(
                        zeile[
                            erste_spalte:
                            letzte_spalte + 1
                        ]
                    ).rstrip()
                )

    # -------------------------
    # STANDORT / LEGENDE
    # -------------------------

    print()
    print("-" * 34)

    print(
        "Standort: "
        + WEGSTATION_RAEUME[
            aktueller_raum
        ]["name"]
    )

    print()
    print("-" * 34)
    print()

    print(
        ">XX< = Dein Standort"
    )

    print(
        "[XX] = Erkundeter Raum"
    )

    print(
        "[?] = Bekannter, noch nicht erkundeter Raum"
    )

    warte_auf_taste(
        "Zurück"
    )
