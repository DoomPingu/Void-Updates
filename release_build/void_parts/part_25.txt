def gegner_schaden_berechnen(starker_angriff):
    if starker_angriff:
        return random.randint(30, 45)

    return random.randint(6, 12)

def gegner_greift_an(name_gegner, lebenspunkte_spieler, schaden_gegner, flucht = False):
    if flucht:
        print(f"{name_gegner} erwischt dich bei der Flucht!")
    else:
        print(f"{name_gegner} greift dich an!")
    lebenspunkte_spieler -= schaden_gegner
    print(f"→ {schaden_gegner} Schaden")

    return lebenspunkte_spieler

def kampf(lebenspunkte_spieler, lebenspunkte_gegner, spiel_laeuft, gold, heiltränke, besiegte_gegner, ort):
    name_gegner = random.choice(gegner_namen)
    starker_angriff = False
    while lebenspunkte_spieler > 0 and lebenspunkte_gegner > 0:
        if not starker_angriff:
            starker_angriff = random.randint(1, 100) <= 20
        print()
        print("=== Kampf ===")
        print(f"{name}: {lebenspunkte_spieler} HP")
        print(f"{name_gegner}: {lebenspunkte_gegner} HP")

        if starker_angriff:
            print()
            print(f"{name_gegner} bereitet einen mächtigen Angriff vor!")
            print(f"{name_gegner} spannt seine Muskeln an und holt weit aus...")
            print()

        print("1. Angreifen")
        print(f"2. Heiltrank benutzen ({heiltränke})")
        print("3. Verteidigen")
        print("4. Fliehen")
        kampf_auswahl = eingabe_menu(
            1,
            4,
            sonder_tasten=()
        )

        schaden_gegner = gegner_schaden_berechnen(starker_angriff)

        if kampf_auswahl == 1:
            schaden_spieler = random.randint(spieler["waffe"]["min_schaden"], spieler["waffe"]["max_schaden"])
            lebenspunkte_gegner = lebenspunkte_gegner - schaden_spieler
            print()
            print(f"Du greifst {name_gegner} an!")
            print(f"→ {schaden_spieler} Schaden")
            time.sleep(1)

            if lebenspunkte_gegner <= 0:
                print(f"{name_gegner} wurde besiegt!")
            else:
                print()
                print(f"{name_gegner} holt zum Angriff aus...")
                time.sleep(1.5)

                lebenspunkte_spieler = gegner_greift_an(
                name_gegner, lebenspunkte_spieler, schaden_gegner)
                time.sleep(1.5)
                starker_angriff = False

        elif kampf_auswahl == 2:
            if heiltränke > 0 and lebenspunkte_spieler < 100:
                lebenspunkte_spieler, heiltränke = heiltrank_benutzen(
                    lebenspunkte_spieler, heiltränke
                )

                gegner_angriff = random.randint(1, 100)

                if gegner_angriff <= 60:
                    time.sleep(1.5)
                    print()

                    lebenspunkte_spieler = gegner_greift_an(
                        name_gegner,
                        lebenspunkte_spieler,
                        schaden_gegner
                    )

                else:
                    print()
                    print(f"{name_gegner} zögert und greift dich nicht an.")

                starker_angriff = False

            else:
                lebenspunkte_spieler, heiltränke = heiltrank_benutzen(
                    lebenspunkte_spieler, heiltränke
        )

        elif kampf_auswahl == 3:

            schaden_gegner = schaden_gegner // 2
            print()
            print("Du gehst in Verteidigungsstellung.")
            print("Der nächste Angriff verursacht nur halben Schaden.")
            time.sleep(1.5)
            print()
            lebenspunkte_spieler = gegner_greift_an(
                name_gegner, lebenspunkte_spieler, schaden_gegner)
            starker_angriff = False

        elif kampf_auswahl == 4:
            fliehen = random.randint(1, 2)
            if fliehen == 1:
                print("Du bist entkommen")
                break
            else:
                print("Entkommen hat nicht geklappt!")
                lebenspunkte_spieler = gegner_greift_an(
                    name_gegner, lebenspunkte_spieler, schaden_gegner, flucht=True)
                starker_angriff = False
    if lebenspunkte_gegner <= 0:
        print()
        print("Du hast gewonnen!")
        gefundenes_gold = random.randint(10, 30)
        gold = gold + gefundenes_gold
        print(f"{name_gegner} hat {gefundenes_gold} Gold fallen gelassen!")
        besiegte_gegner = besiegte_gegner + 1
        if besiegte_gegner == 4:
            print()
            print("Du spürst eine mächtige Präsenz im Wald...")
    elif lebenspunkte_spieler <= 0:
        print("Du wurdest besiegt...")
        print("Game Over")
        spiel_laeuft = False
    return lebenspunkte_spieler, spiel_laeuft, gold, heiltränke, besiegte_gegner, ort

def truhe(gold, heiltränke, diamanten):
    
    truhen_wurf = random.randint(1, 100)
    if truhen_wurf <= 60:
        print()
        print("Du hast eine kleine Truhe gefunden.")
        beute = random.randint(1, 4)
        if beute <= 3:
            gefundenes_gold_klein = random.randint(1, 20)
            print(f"Du findest einen Heiltrank und {gefundenes_gold_klein} Gold!")
            heiltränke = heiltränke + 1
            gold = gold + gefundenes_gold_klein
        elif beute == 4:
            print("Die Truhe ist leer.")
    elif 61 <= truhen_wurf <= 90:
        print()
        print("Du hast eine mittlere Truhe gefunden.")
        beute = random.randint(1, 4)
        if beute <= 3:
            gefundenes_gold_mittel = random.randint(20, 50)
            print(f"Du findest einen Heiltrank und {gefundenes_gold_mittel} Gold!")
            heiltränke = heiltränke + 1
            gold = gold + gefundenes_gold_mittel
        elif beute == 4:
            print("Du findest nichts von Wert.")
    elif 91 <= truhen_wurf <= 100:
        print()
        print("Du hast eine große Truhe gefunden.")
        beute = random.randint(1, 100)
        if beute <= 60:
            gefundenes_gold_gross = random.randint(50, 100)
            print(f"Du findest einen Heiltrank und {gefundenes_gold_gross} Gold!")
            heiltränke = heiltränke + 1
            gold = gold + gefundenes_gold_gross
        elif 61 <= beute <= 85:
            gefundenes_gold_gross = random.randint(50, 100)
            print(f"Du findest ein Diamant und {gefundenes_gold_gross} Gold!")
            diamanten = diamanten + 1
            gold = gold + gefundenes_gold_gross
        else:
            print("Zu deiner Überraschung ist die Truhe leer.")


    return gold, heiltränke, diamanten
