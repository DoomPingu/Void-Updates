
    if slot not in range(
        1,
        ANZAHL_SPEICHER_SLOTS + 1
    ):
        return (
            False,
            ValueError(
                "Ungültiger Speicherplatz."
            )
        )

    speicher_datei = (
        speicherdatei_pfad(slot)
    )

    temp_datei = (
        speicher_datei + ".tmp"
    )

    try:
        with open(
            temp_datei,
            "w",
            encoding="utf-8"
        ) as datei:
            json.dump(
                speicher_daten,
                datei,
                ensure_ascii=False,
                indent=4
            )

        os.replace(
            temp_datei,
            speicher_datei
        )

    except (
        OSError,
        TypeError,
        ValueError
    ) as fehler:

        if os.path.exists(
            temp_datei
        ):
            try:
                os.remove(
                    temp_datei
                )
            except OSError:
                pass

        return False, fehler

    return True, None


def spielstand_backup_erstellen(
    alte_version,
    slot=None
):
    if slot is None:
        slot = AKTIVER_SPEICHER_SLOT

    if slot is None:
        return (
            None,
            ValueError(
                "Kein aktiver Speicherplatz ausgewählt."
            )
        )

    if slot not in range(
        1,
        ANZAHL_SPEICHER_SLOTS + 1
    ):
        return (
            None,
            ValueError(
                "Ungültiger Speicherplatz."
            )
        )

    speicher_datei = (
        speicherdatei_pfad(slot)
    )

    zeitstempel = time.strftime(
        "%Y%m%d-%H%M%S"
    )

    basis, _endung = os.path.splitext(
        speicher_datei
    )

    backup_datei = (
        f"{basis}_backup_v"
        f"{alte_version}_"
        f"{zeitstempel}.json"
    )

    nummer = 2

    while os.path.exists(
        backup_datei
    ):
        backup_datei = (
            f"{basis}_backup_v"
            f"{alte_version}_"
            f"{zeitstempel}_"
            f"{nummer}.json"
        )

        nummer += 1

    try:
        with open(
            speicher_datei,
            "rb"
        ) as quelle:
            inhalt = quelle.read()

        with open(
            backup_datei,
            "xb"
        ) as ziel:
            ziel.write(
                inhalt
            )

    except OSError as fehler:
        return None, fehler

    return backup_datei, None

}


def spielstand_migrieren(
    speicher_daten
):
    version = speicher_daten.get(
        "version",
        1
    )

    spieler_daten = speicher_daten[
        "spieler"
    ]

    while version < SAVE_VERSION:

        migration = SAVE_MIGRATIONEN.get(
            version
        )

        if migration is None:
            raise ValueError(
                "Für diesen Spielstand fehlt "
                "eine notwendige Migration."
            )

        spieler_daten = migration(
            spieler_daten
        )

        version += 1

        speicher_daten[
            "version"
        ] = version

        speicher_daten[
            "spieler"
        ] = spieler_daten

    return speicher_daten

def speicherplatz_text(slot):

        speicherdatei_pfad(slot)
    )

    if not os.path.exists(
        speicher_datei
    ):
        return (
            f"{slot}. Leer"
        )

    try:
        with open(
            speicher_datei,
            "r",
            encoding="utf-8"
        ) as datei:
            speicher_daten = json.load(
                datei
            )

        if not isinstance(
            speicher_daten,
            dict
        ):
            raise ValueError

        spieler_daten = (
            speicher_daten.get(
                "spieler"
            )
        )

        if not isinstance(
            spieler_daten,

        ):
            raise ValueError

            spieler_daten.get(

            )
            or "Unbekannt"
        )

            spieler_daten.get(
                "ort"
            )
            or "Unbekannter Ort"
        )

            f"{slot}. "
