def wegstation_status():

        "wegstation_status",

    status.setdefault(
        "entdeckt",

    )

    status = wegstation_status()

    varianten = status[
        "varianten"
    ]

    if "wachraum" not in varianten:
        varianten[
            "wachraum"
        ] = random.choice(
            [
                "muenzen",

                "frische_spuren",
                "leer"
            ]
        )

    if "schreibstube" not in varianten:
        varianten[
            "schreibstube"
        ] = random.choice(
            [
                "eingestuerztes_regal",

                "persoenliche_notiz",
                "ratten"
            ]
        )

    if "keller" not in varianten:
        varianten[
            "keller"
        ] = random.choice(
            [
                "bandit",
                "wolf",
                "morsche_treppe",
                "frische_spuren"
            ]
        )

    return varianten

def wegstation_raum_betreten(raum_id):
    if raum_id not in WEGSTATION_RAEUME:
        return False

    status = wegstation_status()

    status[
        "aktueller_raum"
    ] = raum_id

    besuchte_raeume = status[
        "besuchte_raeume"
    ]

    erstmals = (
        raum_id not in besuchte_raeume
    )

    if erstmals:
        besuchte_raeume.append(
            raum_id
        )

    return erstmals
