    return besonderer_fund_status(

    ) is not None


def besonderen_fund_dokumentieren(
    fund_id,
    status="gefunden"
):
    if fund_id not in BESONDERE_FUNDE:
        return False

    funde = spieler.setdefault(
        "besondere_funde",
        {}
    )

    alter_status = funde.get(
        fund_id
    )

    funde[fund_id] = status

    return alter_status != status
