def text_taste_pruefen():
    if os.name != "nt":
        return False

    import msvcrt

    enter_gedrueckt = False

    while msvcrt.kbhit():
        taste = msvcrt.getwch()

        # Strg+C weiterhin erlauben
        if taste == "\x03":
            raise KeyboardInterrupt

        # Sondertasten bestehen unter Windows aus zwei Zeichen
        if taste in ("\x00", "\xe0"):
            if msvcrt.kbhit():
                msvcrt.getwch()
            continue

        if taste in ("\r", "\n"):
            enter_gedrueckt = True

    return enter_gedrueckt

TEXTVERLAUF_MAX = 200


def textverlauf_hinzufuegen(text):
    if (
        "spieler" not in globals()
        or not isinstance(spieler, dict)
    ):
        return

    text = str(text).strip()

    if not text:
        return

    verlauf = spieler.setdefault(
        "textverlauf",
        []
    )

    verlauf.append(
        text
    )

    if len(verlauf) > TEXTVERLAUF_MAX:
        del verlauf[
            :-TEXTVERLAUF_MAX
        ]


def textverlauf_anzeigen():
    verlauf = spieler.get(
        "textverlauf",
        []
    )

    if not verlauf:
        ort_titel(
            "Textverlauf"
        )

        print(
            "Noch keine Storytexte vorhanden."
        )

        warte_auf_taste(
            "Zurück"
        )

        return

    eintraege_pro_seite = 8

    letzte_seite = (
        len(verlauf) - 1
    ) // eintraege_pro_seite

    seite = letzte_seite

    while True:
        ort_titel(
            f"Textverlauf - "
            f"Seite {seite + 1}/"
            f"{letzte_seite + 1}"
        )

        start = (
            seite
            * eintraege_pro_seite
        )

        ende = min(
            start
            + eintraege_pro_seite,
            len(verlauf)
        )

        for eintrag in verlauf[
            start:ende
        ]:
            print(
                eintrag
            )

            print()

        menue_trenner()

        optionen = []

        if seite > 0:
            optionen.append(
                (
                    "aelter",
                    "Ältere Texte"
                )
            )

        if seite < letzte_seite:
            optionen.append(
                (
                    "neuer",
                    "Neuere Texte"
                )
            )

        optionen.append(
            (
                "zurueck",
                "Zurück"
            )
        )

        for nummer, (_, text) in enumerate(
            optionen,
            start=1
        ):
            print(
                f"{nummer}. {text}"
            )

        auswahl = eingabe_menu(
            1,
            len(optionen),
            sonder_tasten=()
        )

        aktion = optionen[
            auswahl - 1
        ][0]

        if aktion == "aelter":
            seite -= 1

        elif aktion == "neuer":
            seite += 1

        elif aktion == "zurueck":
            return

def text_ausgeben(text="", art="atmosphaere", stil=None, farbe=None):
    global TEXT_SKIP_AKTIV

    text = spieler_name_formatieren(
        text
    )

    if art == "atmosphaere":
        textverlauf_hinzufuegen(
            text
        )

    if "spieler" in globals() and isinstance(spieler, dict):
        tempo = spieler.get(
            "textgeschwindigkeit",
            STANDARD_EINSTELLUNGEN["textgeschwindigkeit"]
        )
    else:
        tempo = STANDARD_EINSTELLUNGEN["textgeschwindigkeit"]

    faktor = TEXT_FAKTOREN.get(tempo, 1.0)

    stil_start = ""
    stil_ende = ""

    if ANSI_STILE_AKTIV:
        if stil in ANSI_STILE:
            stil_start += ANSI_STILE[stil]

        if farbe in ANSI_STILE:
            stil_start += ANSI_STILE[farbe]

        if stil_start:
            stil_ende = ANSI_STILE["reset"]

    if TEXT_SKIP_AKTIV and art != "atmosphaere":
        warte_auf_taste("Weiter")

    # Wurde ein Storyblock übersprungen,
    # erscheinen alle folgenden Atmosphärentexte
    # bis zur nächsten Bestätigung sofort.
    if TEXT_SKIP_AKTIV and art == "atmosphaere":
        print(
            f"{stil_start}{text}{stil_ende}"
        )
        return

    # Menüs und Status: sofort
    if art == "menue":
        print(f"{stil_start}{text}{stil_ende}")
        return

    # Alle Texteffekte deaktiviert
    if faktor == 0:
        print(f"{stil_start}{text}{stil_ende}")
        return

    # Atmosphäre und Dialoge:
    # Buchstabe für Buchstabe
    if art == "atmosphaere":
        pause = TEXT_BASIS["atmosphaere"] * faktor

        print(stil_start, end="", flush=True)

        position = 0

        while position < len(text):

            if text_taste_pruefen():
                print(
                    text[position:],
                    end="",
                    flush=True
                )

                TEXT_SKIP_AKTIV = True

                position = len(text)

                time.sleep(0.10)
                text_taste_pruefen()

                break

            print(
                text[position],
                end="",
                flush=True
            )

            position += 1
            time.sleep(pause)

        # Eventuelle Tastendrücke während der Ausgabe entfernen
        text_taste_pruefen()

        print(stil_ende)

    # Kampf:
    # Satz sofort, danach kurze Pause
    elif art == "kampf":
        print(f"{stil_start}{text}{stil_ende}")
        time.sleep(TEXT_BASIS["kampf"] * faktor)

    # Wichtige Ereignisse:
    # kurze Pause vorher und längere danach
    elif art == "ereignis":
        time.sleep(0.4 * faktor)
        print(f"{stil_start}{text}{stil_ende}")
        time.sleep(TEXT_BASIS["ereignis"] * faktor)

    else:
        print(f"{stil_start}{text}{stil_ende}")

def basis_ordner():
    if getattr(sys, "frozen", False):
        return os.path.dirname(sys.executable)

    return os.path.dirname(os.path.abspath(__file__))

def design_text(text, design="normal", art="menue"):
    einstellung = TEXT_DESIGN.get(
        design,
        TEXT_DESIGN["normal"]
    )

    text_ausgeben(
        text,
        art=art,
        stil=einstellung["stil"],
        farbe=einstellung["farbe"]
    )

def speicherdatei_pfad(slot):
    return os.path.join(
        basis_ordner(),
        f"void_save_{slot}.json"
    )

def alten_spielstand_importieren():
    if not os.path.exists(
        ALTE_SPEICHER_DATEI
    ):
        return None

    for slot in range(
        1,
        ANZAHL_SPEICHER_SLOTS + 1
    ):
        neue_datei = (
            speicherdatei_pfad(
                slot
            )
        )

        if os.path.exists(
            neue_datei
        ):
            continue

        try:
            os.replace(
                ALTE_SPEICHER_DATEI,
                neue_datei
            )

        except OSError:
            return None

        return slot

    return None

def einstellungen_laden():

        EINSTELLUNGS_DATEI
    ):
        return False

    try:
        with open(
            EINSTELLUNGS_DATEI,
            "r",
            encoding="utf-8"
        ) as datei:
            daten = json.load(
                datei
            )

            daten,
            dict
        ):
            return False

    except (
        OSError,
        json.JSONDecodeError
    ):
