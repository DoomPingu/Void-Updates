        "funde"
    ]

    aktionen = []

    if (
        raum_id == "wachraum"
        and not funde.get(

            False
        )
    ):
        aktionen.append(
            (
                "wachraum_untersuchen",
                "Den Wachraum durchsuchen"
            )
        )

    if (
        raum_id == "schreibstube"
        and not funde.get(
            "stationsregister_gelesen",
            False
        )
    ):
        aktionen.append(
            (
                "schreibstube_untersuchen",
                "Die alten Unterlagen untersuchen"
            )
        )

    if (
        raum_id == "keller"
        and not funde.get(
            "keller_untersucht",
            False
        )
    ):
        aktionen.append(
            (
                "keller_untersuchen",
                "Den hinteren Keller untersuchen"
            )
        )

    elif (
        raum_id == "keller"
        and "keller_vorhof"
        not in status["abkuerzungen"]
    ):
        aktionen.append(
            (
                "keller_abkuerzung",
                "Die alte Seitentür untersuchen"
            )
        )

    if (
        raum_id == "archivkammer"
        and not funde.get(
            "archiv_untersucht",
            False
        )
    ):
        aktionen.append(
            (
                "archiv_untersuchen",
                "Die alten Karten und Register untersuchen"
            )
        )

    elif (
        raum_id == "archivkammer"
        and not funde.get(
            "geheimraum_hinweis",
            False
        )
        and not status.get(
            "geheimgang_offen",

        )

                "altes_kartenstueck"

        raum_id == "archivkammer"
        and funde.get(
            "geheimraum_hinweis",

        )

            False

        raum_id == "versiegelte_kammer"
        and not funde.get(
            "versiegeltes_register_geborgen",
            False
        )
    ):
        aktionen.append(
            (
                "versiegeltes_register",
                "Den steinernen Sockel untersuchen"
            )
        )

    return aktionen

def wegstation_verbindungen(raum_id):
    raum = WEGSTATION_RAEUME.get(
        raum_id
    )

    if not raum:
        return []

    verbindungen = list(
        raum.get(
            "wege",
            []
        )
    )

    status = wegstation_status()

    if (
        raum_id == "keller"
        and not status["funde"].get(
            "keller_untersucht",
            False
        )
        and "archivkammer" in verbindungen
    ):
        verbindungen.remove(
            "archivkammer"
        )

    # Spätere Abkürzung:
    # Keller <-> Vorhof
    if (
        "keller_vorhof"
        in status["abkuerzungen"]
    ):
        if (
            raum_id == "keller"
            and "vorhof" not in verbindungen
        ):
            verbindungen.append(
                "vorhof"
            )

        elif (
            raum_id == "vorhof"
            and "keller" not in verbindungen
        ):
            verbindungen.append(
                "keller"
            )

    # Späterer Geheimgang
    if status.get(
        "geheimgang_offen",
        False
    ):
        if (
            raum_id == "archivkammer"
            and "versiegelte_kammer"
            not in verbindungen
        ):
            verbindungen.append(
                "versiegelte_kammer"
            )

    return verbindungen


def wegstation_gebietskarte_anzeigen():
    daten = wegstation_status()

    aktueller_raum = daten.get(
        "aktueller_raum",
        "vorhof"
    )

    besuchte_raeume = set(
        daten.get(
            "besuchte_raeume",
            []
        )
    )

    # -------------------------
    # SICHTBARKEIT
    # -------------------------

    def raum_sichtbar(raum_id):
        if raum_id in besuchte_raeume:
            return True

        for besucht_id in besuchte_raeume:
            if raum_id in wegstation_verbindungen(
                besucht_id
            ):
                return True

        return False

    # -------------------------
    # KARTENFELD
    # -------------------------

    def feld(raum_id):
        if not raum_sichtbar(
            raum_id
        ):
            return None

        if raum_id not in besuchte_raeume:
            return "[?]"

        name = WEGSTATION_KARTENNAMEN[
            raum_id
        ]

        if raum_id == aktueller_raum:
            return f">{name}<"

        return f"[{name}]"

    # -------------------------
    # ZEICHENBEREICH
    # -------------------------

    breite = 56
    hoehe = 9

    karte = [
        [" "] * breite
        for _ in range(hoehe)
    ]

    # -------------------------
    # VERBINDUNGSLINIEN
    # -------------------------

    def mittelpunkt(raum_id):
        text = feld(
            raum_id
        )

        if text is None:
            return None

        zeile, spalte = (
            WEGSTATION_KARTENPOSITIONEN[
                raum_id
            ]
        )

        return (
            zeile,
            spalte + len(text) // 2
        )

    def linie_setzen(
        zeile,
        spalte,
        zeichen
    ):
        if not (
            0 <= zeile < hoehe
            and 0 <= spalte < breite
        ):
            return

        if karte[zeile][spalte] == " ":
            karte[zeile][spalte] = zeichen

    def verbindung_zeichnen(
        start_id,
        ziel_id
    ):
        start = mittelpunkt(
            start_id
        )

        ziel = mittelpunkt(
            ziel_id
        )

        if (
            start is None
            or ziel is None
        ):
            return

        start_zeile, start_spalte = start
        ziel_zeile, ziel_spalte = ziel

        # Gleiche Höhe
        if start_zeile == ziel_zeile:
            von = min(
                start_spalte,
                ziel_spalte
            )

            bis = max(
                start_spalte,
                ziel_spalte
            )

            for spalte in range(
                von + 1,
                bis
            ):
                linie_setzen(
                    start_zeile,
                    spalte,
                    "─"
                )

            return

        # Gleiche Spalte
        if start_spalte == ziel_spalte:
            von = min(
                start_zeile,
                ziel_zeile
            )

            bis = max(
                start_zeile,
                ziel_zeile
            )

            for zeile in range(
                von + 1,
                bis
            ):
                linie_setzen(
                    zeile,
                    start_spalte,
                    "│"
                )

            return

        # Unterschiedliche Höhe und Spalte
        schritt = (
            1
            if ziel_spalte > start_spalte
            else -1
        )

        for spalte in range(
            start_spalte + schritt,
            ziel_spalte,
            schritt
        ):
            linie_setzen(
                start_zeile,
                spalte,
                "─"
            )

        schritt = (
            1
            if ziel_zeile > start_zeile
            else -1
        )

        for zeile in range(
            start_zeile + schritt,
            ziel_zeile,
            schritt
        ):
            linie_setzen(
                zeile,
                ziel_spalte,
                "│"
            )

    # -------------------------
    # SICHTBARE VERBINDUNGEN
    # -------------------------

    gezeichnete_verbindungen = set()

    for start_id in besuchte_raeume:
        for ziel_id in wegstation_verbindungen(
            start_id
        ):
            if (
                start_id
                not in WEGSTATION_KARTENPOSITIONEN
                or ziel_id
                not in WEGSTATION_KARTENPOSITIONEN
            ):
                continue

            if not raum_sichtbar(
                ziel_id
            ):
                continue

            verbindung = frozenset(
                (
                    start_id,
                    ziel_id
                )
            )

            if verbindung in gezeichnete_verbindungen:
                continue

            gezeichnete_verbindungen.add(
                verbindung
            )

            verbindung_zeichnen(
                start_id,
                ziel_id
            )

    # -------------------------
    # RÄUME EINZEICHNEN
    # -------------------------

    for (
        raum_id,
        position
    ) in WEGSTATION_KARTENPOSITIONEN.items():

        text = feld(
            raum_id
        )

        if text is None:
            continue

        zeile, spalte = position

        for index, zeichen in enumerate(
            text
        ):
            x = spalte + index

            if (
                0 <= zeile < hoehe
                and 0 <= x < breite
            ):
                karte[
                    zeile
                ][x] = zeichen

    # -------------------------
    # KARTE AUSGEBEN
    # -------------------------

    ort_titel(
        "Gebietskarte - Alte Wegstation"
    )

    sichtbare_zeilen = [
        index
        for index, zeile in enumerate(karte)
        if any(
            zeichen != " "
            for zeichen in zeile
        )
    ]

    if sichtbare_zeilen:
        erste_zeile = min(
            sichtbare_zeilen
        )

        letzte_zeile = max(
            sichtbare_zeilen
        )

        sichtbarer_bereich = karte[
            erste_zeile:
            letzte_zeile + 1
        ]

        belegte_spalten = []

        for zeile in sichtbarer_bereich:
            for index, zeichen in enumerate(
                zeile
            ):
                if zeichen != " ":
                    belegte_spalten.append(
                        index
                    )

        if belegte_spalten:
            erste_spalte = min(
                belegte_spalten
            )

            letzte_spalte = max(
                belegte_spalten
            )

            for zeile in sichtbarer_bereich:
                print(
                    "  "
                    + "".join(
                        zeile[
                            erste_spalte:
                            letzte_spalte + 1
                        ]
                    ).rstrip()
                )

    # -------------------------
    # STANDORT / LEGENDE
    # -------------------------

    print()
    print("-" * 34)

    print(
        "Standort: "
        + WEGSTATION_RAEUME[
            aktueller_raum
        ]["name"]
    )

    print()
    print("-" * 34)
    print()

    print(
        ">XX< = Dein Standort"
    )

    print(
        "[XX] = Erkundeter Raum"
    )

    print(
        "[?] = Bekannter, noch nicht erkundeter Raum"
    )

    warte_auf_taste(
        "Zurück"
    )

def alte_wegstation():
    dungeon_status = wegstation_status()

    if not dungeon_status[
        "entdeckt"
    ]:
        dungeon_status[
            "entdeckt"
        ] = True

    raum_id = dungeon_status.get(
        "aktueller_raum",
        "vorhof"
    )

    if raum_id not in WEGSTATION_RAEUME:
        raum_id = "vorhof"

    erstmals = wegstation_raum_betreten(
        raum_id
    )

    raum = WEGSTATION_RAEUME[
        raum_id
    ]

    ort_titel(
        f"Alte Wegstation - {raum['name']}"
    )

    ort_nachricht_anzeigen()

    if erstmals:
        for zeile in raum.get(
            "beschreibung",
            []
        ):
            text_ausgeben(
                zeile
            )

            print()

        if raum_id == "vorhof":
            design_text(
                "KÖNIGLICHE WEGSTATION",
                design="wissen",
                art="menue"
            )

            print()

            text_ausgeben(
                "Ein zweiter Schriftzug darunter ist fast "
                "vollständig verwittert."
            )

    else:
        print(
            raum.get(
                "kurztext",
                "Du siehst dich erneut um."
            )
        )

    wegstation_varianten_initialisieren()

    verbindungen = wegstation_verbindungen(
        raum_id
    )

    aktionen = wegstation_raum_aktionen(
        raum_id
    )

    menue_eintraege = []

    for ziel_id in verbindungen:
        menue_eintraege.append(
            (
                "weg",
                ziel_id,
                WEGSTATION_RAEUME[
                    ziel_id
                ]["name"]
            )
        )

    for (
        aktion_id,
        aktion_text
    ) in aktionen:
        menue_eintraege.append(
            (
                "aktion",
                aktion_id,
                aktion_text
            )
        )

    if raum_id == "vorhof":
        menue_eintraege.append(
            (
                "verlassen",
                None,
                "Zum Königsring zurückkehren"
            )
        )

    menue_trenner()

    for nummer, (
        eintrag_typ,
        eintrag_id,
        eintrag_text
    ) in enumerate(
        menue_eintraege,
        start=1
    ):
        print(
            f"{nummer}. "
            f"{eintrag_text}"
        )

    print()
    print("[K] Gebietskarte")
    print("[S] Status")
    print("[I] Inventar")
    print("[J] Questlog/Erfolge")
    print("[Q] Speichern / Menü")

    auswahl = eingabe_menu(
        1,
        len(menue_eintraege),
        sonder_tasten=(
            "k",
            "s",
            "i",
            "j",
            "q"
        )
    )

    if auswahl == "k":
        wegstation_gebietskarte_anzeigen()
        return
    
    if auswahl == "s":
        status()
        return

    if auswahl == "i":
        inventar()
        return

    if auswahl == "j":
        questlog()
        return

    if auswahl == "q":
        spielmenue()
        return

    (
        eintrag_typ,
        eintrag_id,
        eintrag_text
    ) = menue_eintraege[
        auswahl - 1
    ]

    if eintrag_typ == "verlassen":
        aktive_reise = spieler.get(
            "reise"
        )

        if isinstance(
            aktive_reise,
            dict
        ):
            spieler["ort"] = aktive_reise.get(
                "gebiet",
                "Wald"
            )

            ort_nachricht_setzen(
                "Du lässt die alte Wegstation hinter dir.",
                "Wenig später stehst du wieder auf dem Königsring."
            )

        else:
            spieler["ort"] = "Eichenruh"

            ort_nachricht_setzen(
                "Du lässt die alte Wegstation hinter dir.",
                "Nach einiger Zeit erreichst du wieder Eichenruh."
            )

        dungeon_status[
            "aktueller_raum"
        ] = "vorhof"

        return

    if eintrag_typ == "weg":
        dungeon_status[
            "aktueller_raum"
        ] = eintrag_id

        return

    if eintrag_typ == "aktion":

        if (
            eintrag_id
            == "wachraum_untersuchen"
        ):
            wegstation_wachraum_untersuchen()

        elif (
            eintrag_id
            == "schreibstube_untersuchen"
        ):
            wegstation_schreibstube_untersuchen()

        elif (
            eintrag_id
            == "keller_untersuchen"
        ):
            wegstation_keller_untersuchen()

        elif (
            eintrag_id
            == "keller_abkuerzung"
        ):
            wegstation_keller_abkuerzung_oeffnen()

        elif (
            eintrag_id
            == "archiv_untersuchen"
        ):
            wegstation_archiv_untersuchen()


        elif (
            eintrag_id
            == "kartenstueck_archiv"
        ):
            wegstation_kartenstueck_archiv_vergleichen()

        elif (
            eintrag_id
            == "geheimraum_oeffnen"
        ):
            wegstation_geheimraum_oeffnen()
