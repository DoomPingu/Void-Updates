def warte_auf_taste(text="Zurück"):
    global TEXT_SKIP_AKTIV

    if os.name == "nt":
        import msvcrt

        # Alte Tastendrücke entfernen.
        # Eine Taste, die bereits VOR dieser Bestätigung
        # gedrückt wurde, darf nicht als Weiter zählen.
        while msvcrt.kbhit():
            taste = msvcrt.getwch()

            if taste == "\x03":
                raise KeyboardInterrupt

            if taste in ("\x00", "\xe0"):
                if msvcrt.kbhit():
                    msvcrt.getwch()

    print()
    print(f"[Beliebige Taste] {text}")

    if os.name == "nt":
        import msvcrt

        while True:
            taste = msvcrt.getwch()

            if taste == "\x03":
                raise KeyboardInterrupt

            # Pfeiltasten / Sondertasten überspringen
            if taste in ("\x00", "\xe0"):
                if msvcrt.kbhit():
                    msvcrt.getwch()
                continue

            TEXT_SKIP_AKTIV = False
            return

    else:
        input()
        TEXT_SKIP_AKTIV = False

def eingabe_zahl(minimum, maximum, trenner=True):
    fehler = False

    while True:
        if fehler and ANSI_STILE_AKTIV:
            print(
                "\033[F\033[2K",
                end="",
                flush=True
            )

        if fehler:
            prompt = (
                f"Ungültig – wähle "
                f"{minimum}-{maximum} > "
            )
        else:
            prompt = "Auswahl: "

        eingabe = input(
            prompt
        ).strip()

        try:
            zahl = int(
                eingabe
            )
        except ValueError:
            zahl = None

        if (
            zahl is not None
            and minimum <= zahl <= maximum
        ):
            if fehler and ANSI_STILE_AKTIV:
                print(
                    "\033[F\033[2K",
                    end="",
                    flush=True
                )

                print(
                    f"Auswahl: {zahl}"
                )

            if trenner:
                auswahl_trenner()

            return zahl

        fehler = True

def eingabe_menu(
    minimum,
    maximum,
    sonder_tasten=("s", "i"),
    trenner=True
):
    sonder_tasten = tuple(
        taste.lower()
        for taste in sonder_tasten
    )

    if isinstance(
        spieler,
        dict
    ):
        eingabe_modus = spieler.get(
            "menue_eingabe",
            STANDARD_EINSTELLUNGEN[
                "menue_eingabe"
            ]
        )

    else:
        eingabe_modus = (
            STANDARD_EINSTELLUNGEN[
                "menue_eingabe"
            ]
        )

    # -------------------------
    # DIREKTE TASTENEINGABE
    # -------------------------

    if (
        eingabe_modus == "direkt"
        and os.name == "nt"
        and maximum <= 9
    ):
        import msvcrt

        print(
            "Auswahl: ",
            end="",
            flush=True
        )

        while True:
            taste = (
                msvcrt.getwch()
                .lower()
            )

            if taste == "\x03":
                raise KeyboardInterrupt

            # Pfeiltasten / Sondertasten
            if taste in (
                "\x00",
                "\xe0"
            ):
                msvcrt.getwch()
                continue

            # S, I, J, Q usw.
            if taste in sonder_tasten:
                print(
                    taste.upper()
                )

                if trenner:
                    auswahl_trenner()

                return taste

            # Zahlen
            if taste.isdigit():
                zahl = int(
                    taste
                )

                if (
                    minimum
                    <= zahl
                    <= maximum
                ):
                    print(
                        taste
                    )

                    if trenner:
                        auswahl_trenner()

                    return zahl

    # -------------------------
    # MIT ENTER BESTÄTIGEN
    # -------------------------

    fehler = False

    while True:
        if (
            fehler
            and ANSI_STILE_AKTIV
        ):
            print(
                "\033[F\033[2K",
                end="",
                flush=True
            )

        if fehler:
            prompt = (
                f"Ungültig – wähle "
                f"{minimum}-{maximum} > "
            )

        else:
            prompt = "Auswahl: "

        eingabe = input(
            prompt
        ).strip().lower()

        # -------------------------
        # SONdertasten
        # -------------------------

        if eingabe in sonder_tasten:
            if (
                fehler
                and ANSI_STILE_AKTIV
            ):
                print(
                    "\033[F\033[2K",
                    end="",
                    flush=True
                )

                print(
                    f"Auswahl: "
                    f"{eingabe.upper()}"
                )

            if trenner:
                auswahl_trenner()

            return eingabe

        # -------------------------
        # ZAHL
        # -------------------------

        try:
            zahl = int(
                eingabe
            )

        except ValueError:
            zahl = None

        if (
            zahl is not None
            and minimum <= zahl <= maximum
        ):
            if (
                fehler
                and ANSI_STILE_AKTIV
            ):
                print(
                    "\033[F\033[2K",
                    end="",
                    flush=True
                )

                print(
                    f"Auswahl: {zahl}"
                )

            if trenner:
                auswahl_trenner()

            return zahl

        fehler = True

    ):

        )

        heiltränke -= 1

        print("Du hast keine Heiltränke!")

    elif lebenspunkte == 100:
        print(

        )

    return lebenspunkte, heiltränke
