def haendler_kaufen(
    titel,
    begruessung,
    sortiment
):
    while True:
        ort_titel(titel)

        print(
            begruessung
        )

        print()

        print(
            f"Gold vorhanden: {spieler['gold']}"
        )

        print()

        waren = list(
            sortiment.items()
        )

        for nummer, (
            item_id,
            handels_daten
        ) in enumerate(
            waren,
            start=1
        ):
            print(
                f"{nummer}. "
                f"{ITEMS[item_id]['name']} "
                f"- {handels_daten['preis']} Gold "
                f"(Du hast: {item_anzahl(item_id)})"
            )

        zurueck_nummer = (
            len(waren) + 1
        )

        print()
        print(
            f"{zurueck_nummer}. Zurück"
        )

        auswahl = eingabe_menu(
            1,
            zurueck_nummer,
            sonder_tasten=()
        )

        if auswahl == zurueck_nummer:
            return

        item_id, handels_daten = (
            waren[
                auswahl - 1
            ]
        )

        preis = handels_daten[
            "preis"
        ]

        max_anzahl = (
            spieler["gold"] // preis
        )

        if max_anzahl <= 0:
            print()
            print(
                "Du hast nicht genug Gold."
            )

            warte_auf_taste()
            continue

        ort_titel(
            f"{ITEMS[item_id]['name']} kaufen"
        )

        print(
            f"Preis: {preis} Gold pro Stück"
        )

        print(
            f"Gold vorhanden: {spieler['gold']}"
        )

        print(
            f"Du kannst maximal "
            f"{max_anzahl} kaufen."
        )

        print()
        print("0. Zurück")

        anzahl = eingabe_zahl(
            0,
            max_anzahl
        )

        if anzahl == 0:
            continue

        gesamtkosten = (
            anzahl * preis
        )

        spieler[
            "gold"
        ] -= gesamtkosten

        item_hinzufuegen(
            item_id,
            anzahl
        )

        print()

        design_text(
            (
                f"{anzahl}x "
                f"{ITEMS[item_id]['name']} erhalten"
            ),
            design="positiv",
            art="menue"
        )

        print(
            f"Bezahlt: "
            f"{gesamtkosten} Gold"
        )

        print(
            f"Verbleibendes Gold: "
            f"{spieler['gold']}"
        )

        print(
            f"{ITEMS[item_id]['name']}: "
            f"{item_anzahl(item_id)}"
        )

        warte_auf_taste()
