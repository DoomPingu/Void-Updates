            ]
        )
    }

    temp_datei = (
        EINSTELLUNGS_DATEI + ".tmp"
    )

    try:
        with open(
            temp_datei,
            "w",
            encoding="utf-8"
        ) as datei:
            json.dump(
                daten,
                datei,
                ensure_ascii=False,
                indent=4
            )

            temp_datei,
            EINSTELLUNGS_DATEI
        )

    except (
        OSError,
        TypeError,
        ValueError
    ):
        if os.path.exists(
            temp_datei
        ):
            try:
                os.remove(
                    temp_datei
                )
            except OSError:
                pass

        return False

    return True

def spieler_name_formatieren(text):
    if not isinstance(
        globals().get("spieler"),
        dict
    ):
        return text

    name = spieler.get("name")

    if not name:
        return text

    name_text = f"[{name}]"

    if not text.startswith(name_text):
        return text

    farbe = spieler.get(
        "spieler_name_farbe",
        STANDARD_EINSTELLUNGEN[
            "spieler_name_farbe"
        ]
    )

    if (
        farbe == "standard"
        or not ANSI_STILE_AKTIV
        or farbe not in ANSI_STILE
    ):
        return text

    return (
        f"{ANSI_STILE[farbe]}"
        f"{name_text}"
        f"{ANSI_STILE['reset']}"
        f"{text[len(name_text):]}"
    )

def reise(
    ziel,
    *ankunft_zeilen,
    gebiet_modus=False
):
    start = spieler["ort"]
    strecke = frozenset((start, ziel))

    if strecke not in REISE_STRECKEN:
        print(
            "Für diese Strecke ist noch keine Verbindung festgelegt."
        )
        warte_auf_taste()
        return

    reise_daten = REISE_STRECKEN[
        strecke
    ]

    weg_id = reise_daten.get(
        "weg_id"
    )

    # -------------------------
    # VERTRAUTE ROUTE
    # -------------------------

    if (
        weg_id
        and reise_daten.get(
            "schnellreise",
            False
        )
        and routenkenntnis_bekannt(
            weg_id
        )
    ):
        reise_name = reise_daten.get(
            "name",
            reise_daten["gebiet"]
        )

        ort_titel(
            f"Reise - {reise_name}"
        )

        print(
            f"{start} → {ziel}"
        )

        print()

        design_text(
            "Vertraute Route",
            design="wissen",
            art="menue"
        )

        print()

        print(
            "Du kennst diese Strecke inzwischen gut genug, "
            "um sie ohne längere Erkundung zurückzulegen."
        )

        menue_trenner()

        print(
            "1. Normal reisen"
        )

        print(
            "2. Strecke zügig zurücklegen"
        )

        print(
            "3. Zurück"
        )

        reise_auswahl = eingabe_menu(
            1,
            3,
            sonder_tasten=()
        )

        if reise_auswahl == 3:
            return

        # -------------------------
        # ZÜGIG REISEN
        # -------------------------

        if reise_auswahl == 2:

            # Zeit vergeht trotzdem.
            if reise_daten[
                "gebiet"
            ] in (
                "Wald",
                "Waldweg"
            ):
                ressourcen_vorkommen_regenerieren(
                    "wald_holz",
                    reise_daten[
                        "schritte"
                    ]
                )

            if not status_schritte_abschliessen(
                reise_daten[
                    "schritte"
                ]
            ):
                return

            spieler[
                "reise"
            ] = None

            spieler[
                "ort"
            ] = ziel

            ort_nachricht_setzen(
                (
                    f"Du folgst den vertrauten Markierungen auf dem "
                    f"{reise_name} und vermeidest unnötige Umwege."
                ),
                *ankunft_zeilen
            )

            return

    # -------------------------
    # NORMALE REISE
    # -------------------------

    spieler["reise"] = {
        "start": start,
        "ziel": ziel,
        "gebiet": reise_daten[
            "gebiet"
        ],

        "name": reise_daten.get(
            "name",
            reise_daten["gebiet"]
        ),

        "weg_id": reise_daten.get(
            "weg_id"
        ),

        "gesamt_schritte": reise_daten[
            "schritte"
        ],

        "aktueller_schritt": 0,

        "ausgeloeste_ereignisse": [],

        "angebotene_abzweigungen": [],

        "zufallsereignisse": reise_daten.get(
            "zufallsereignisse",
            True
        ),

        "ankunft_zeilen": list(
            ankunft_zeilen
        )
    }

    if gebiet_modus:
        spieler[
            "ort"
        ] = reise_daten[
            "gebiet"
        ]

        ort_nachricht_setzen(
            f"Du verlässt {start} und betrittst den "
            f"{reise_daten['gebiet']}.",
            f"Der Weg führt weiter in Richtung {ziel}."
        )

        return

    reise_fortsetzen()

def status_hinzufuegen(
    status_id,
    dauer=None,
    einheit=None,
    stapel=1,
    quelle=None,
    meldung=True
):
    definition = STATUS_EFFEKTE.get(
        status_id
    )

    if not definition:
        return False

    if dauer is None:
        dauer = definition.get(
            "standard_dauer"
        )

    if einheit is None:
        einheit = definition.get(
            "standard_einheit"
        )

    if (
        dauer is not None
        and (
            isinstance(dauer, bool)
            or not isinstance(
                dauer,
                (int, float)
            )
            or dauer <= 0
        )
    ):
        return False

    if (
        isinstance(stapel, bool)
        or not isinstance(stapel, int)
        or stapel <= 0
    ):
        return False

    aktive = status_effekte_aktiv()

    vorhanden = aktive.get(
        status_id
    )

    if vorhanden:
        if definition.get(
            "stapelbar",
            False
        ):
            max_stapel = definition.get(
                "max_stapel",
                1
            )

            vorhanden["stapel"] = min(
                max_stapel,
                vorhanden.get(
                    "stapel",
                    1
                ) + stapel
            )

        if dauer is not None:
            if (
                vorhanden.get(
                    "einheit"
                ) == einheit
                and isinstance(
                    vorhanden.get(
                        "dauer"
                    ),
                    (int, float)
                )
            ):
                vorhanden["dauer"] = max(
                    vorhanden["dauer"],
                    dauer
                )

            else:
                vorhanden[
                    "dauer"
                ] = dauer

                vorhanden[
                    "einheit"
                ] = einheit

        if quelle is not None:
            vorhanden[
                "quelle"
            ] = quelle

    else:
        aktive[status_id] = {
            "dauer": dauer,
            "einheit": einheit,
            "stapel": min(
                stapel,
                definition.get(
                    "max_stapel",
                    stapel
                )
                if definition.get(
                    "stapelbar",
                    False
                )
                else 1
            ),
            "quelle": quelle
        }

    if meldung:
        print()

        art = definition.get(
            "art",
            "zustand"
        )

        if art == "buff":
            titel = "BUFF AKTIV"

        elif art == "debuff":
            titel = "DEBUFF AKTIV"

        else:
            titel = "STATUS AKTIV"

        design_text(
            titel,
            design=definition.get(
                "design",
                "wissen"
            ),
            art="menue"
        )

        print(
            definition["name"]
        )

    return True


def status_entfernen(
    status_id,
    meldung=True
):
    aktive = status_effekte_aktiv()

    if status_id not in aktive:

    aktive.pop(

    if meldung:
        definition = STATUS_EFFEKTE.get(
            status_id,

        print()

            "STATUS BEENDET",
            design="wissen",
            art="menue"
        )

        print(
            definition.get(
                "name",
                status_id
            )
        )

    return True
