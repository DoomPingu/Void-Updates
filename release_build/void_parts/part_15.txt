def rezept_fehlende_zutaten(rezept):
    fehlende_zutaten = []

    for item_id, anzahl in rezept[
        "zutaten"
    ].items():

        vorhanden = item_anzahl(
            item_id
        )

        if vorhanden < anzahl:
            fehlende_zutaten.append(
                (
                    item_id,
                    anzahl - vorhanden
                )
            )

    return fehlende_zutaten


def rezept_kann_gekocht_werden(rezept):
    return not rezept_fehlende_zutaten(
        rezept
    )


def kochen(titel="Kochen"):
    while True:
        ort_titel(titel)

        gelernte_rezepte = [
            rezept_id
            for rezept_id in spieler.get(
                "rezepte_gelernt",
                []
            )
            if rezept_id in REZEPTE
        ]

        if not gelernte_rezepte:
            print(
                "Du kennst momentan keine Rezepte."
            )

            print()
            print(
                "Vielleicht findest du irgendwo "
                "ein Kochbuch."
            )

            warte_auf_taste()
            return

        print("Bekannte Rezepte:")
        print()

        for nummer, rezept_id in enumerate(
            gelernte_rezepte,
            start=1
        ):
            rezept = REZEPTE[
                rezept_id
            ]

            ergebnis_id = rezept[
                "ergebnis"
            ]

            heilung = ITEMS[
                ergebnis_id
            ].get(
                "heilung",
                0
            )

            fehlende_zutaten = (
                rezept_fehlende_zutaten(
                    rezept
                )
            )

            print(
                f"{nummer}. "
                f"{rezept['name']} "
                f"- {heilung} HP"
            )

            if not fehlende_zutaten:
                design_text(
                    "   [Bereit]",
                    design="positiv",
                    art="menue"
                )

            else:
                fehlende_texte = []

                for (
                    item_id,
                    anzahl
                ) in fehlende_zutaten:

                    fehlende_texte.append(
                        f"{ITEMS[item_id]['name']} "
                        f"x{anzahl}"
                    )

                print(
                    "   [Fehlt: "
                    + ", ".join(
                        fehlende_texte
                    )
                    + "]"
                )

            print()

        zurueck_nummer = (
            len(gelernte_rezepte) + 1
        )

        print()
        print(
            f"{zurueck_nummer}. Zurück"
        )

        auswahl = eingabe_menu(
            1,
            zurueck_nummer,
            sonder_tasten=()
        )

        if auswahl == zurueck_nummer:
            return

        rezept_id = gelernte_rezepte[
            auswahl - 1
        ]

        rezept = REZEPTE[
            rezept_id
        ]

        ort_titel(
            rezept["name"]
        )

        ergebnis_id = rezept[
            "ergebnis"
        ]

        heilung = ITEMS[
            ergebnis_id
        ].get(
            "heilung",
            0
        )

        print(
            f"Heilung: {heilung} HP"
        )

        print()
        print("Benötigte Zutaten:")
        print()

        for item_id, anzahl in rezept[
            "zutaten"
        ].items():

            vorhanden = item_anzahl(
                item_id
            )

            if vorhanden >= anzahl:
                status = "OK"

            else:
                fehlt = (
                    anzahl - vorhanden
                )

                status = (
                    f"Fehlt: {fehlt}"
                )

            print(
                f"- {ITEMS[item_id]['name']} "
                f"x{anzahl} "
                f"(Du hast: {vorhanden}) "
                f"[{status}]"
            )

        print()

        if not rezept_kann_gekocht_werden(
            rezept
        ):
            print(
                "Dir fehlen Zutaten für "
                "dieses Rezept."
            )

            warte_auf_taste()
            continue

        print("1. Zubereiten")
        print("2. Zurück")

        koch_auswahl = eingabe_menu(
            1,
            2,
            sonder_tasten=()
        )

        if koch_auswahl == 2:
            continue

        for item_id, anzahl in rezept[
            "zutaten"
        ].items():

            item_entfernen(
                item_id,
                anzahl
            )

        item_hinzufuegen(
            rezept["ergebnis"],
            rezept.get(
                "anzahl",
                1
            )
        )

        print()

        design_text(
            f"{rezept['name']} zubereitet",
            design="positiv",
            art="menue"
        )

        warte_auf_taste()
